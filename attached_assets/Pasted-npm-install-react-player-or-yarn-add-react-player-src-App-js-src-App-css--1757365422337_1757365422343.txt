import React, { useState } from "react";
import ReactPlayer from "react-player";
import "./App.css";

const SAMPLE_VIDEOS = [
  { id: "1", title: "Video 1 - Intro", url: "https://www.youtube.com/watch?v=dQw4w9WgXcQ" },
  { id: "2", title: "Video 2 - Demo", url: "https://www.youtube.com/watch?v=3JZ_D3ELwOQ" },
  { id: "3", title: "Video 3 - Tutorial", url: "https://www.youtube.com/watch?v=L_jWHffIx5E" }
];

function BMOButton({ onStart }) {
  return (
    <div className="bmo-button" onClick={onStart} role="button" tabIndex={0}>
      <div className="bmo-body">
        <div className="bmo-screen-small">
          <div className="eyes">
            <div className="eye" />
            <div className="eye" />
          </div>
        </div>
        <div className="bmo-controls">
          <div className="btn red" />
          <div className="btn blue" />
          <div className="dpad" />
          <div className="btn green" />
        </div>
        <div className="bmo-legs">
          <div className="leg"><div className="shoe"/></div>
          <div className="leg"><div className="shoe"/></div>
        </div>
      </div>
      <div className="start-hint">Click BMO to Start</div>
    </div>
  );
}

function VideoExplorer({ videos, onSelect }) {
  return (
    <div className="video-explorer">
      <div className="explorer-toolbar">C:\Portfolio\Videos</div>
      <div className="explorer-content">
        <div className="video-list">
          {videos.map(v => (
            <div key={v.id} className="video-item" onClick={() => onSelect(v)}>
              <div className="thumb">{/* could put thumbnail */}</div>
              <div className="v-title">{v.title}</div>
            </div>
          ))}
        </div>
        <div className="player-placeholder">Select a video to play</div>
      </div>
    </div>
  );
}

function App() {
  const [started, setStarted] = useState(false);
  const [activeSection, setActiveSection] = useState(null); // "videos" | null
  const [currentVideo, setCurrentVideo] = useState(null);

  return (
    <div className="app">
      {!started ? (
        <div className="landing">
          <BMOButton onStart={() => setStarted(true)} />
        </div>
      ) : (
        <div className="workspace">
          <div className="bmo-expanded">
            <div className="bmo-screen-expanded">
              <div className="screen-toolbar">
                <button className={`tab ${activeSection === null ? "active":""}`} onClick={() => { setActiveSection(null); setCurrentVideo(null); }}>Home</button>
                <button className={`tab ${activeSection === "videos" ? "active":""}`} onClick={() => { setActiveSection("videos"); setCurrentVideo(null); }}>Videos</button>
                <button className={`tab`} onClick={() => { setActiveSection("ai"); setCurrentVideo(null); }}>AI Talk</button>
                <div className="path">C:\Portfolio\{activeSection ? activeSection : ""}</div>
              </div>

              <div className="screen-body">
                {!activeSection && (
                  <div className="grid-icons">
                    <div className="icon-card" onClick={() => setActiveSection("videos")}>
                      <div className="icon">🎬</div>
                      <div className="label">Videos</div>
                    </div>
                    <div className="icon-card" onClick={() => setActiveSection("ai")}>
                      <div className="icon">💬</div>
                      <div className="label">AI Talk</div>
                    </div>
                    <div className="icon-card">
                      <div className="icon">👥</div>
                      <div className="label">Communities</div>
                    </div>
                    <div className="icon-card">
                      <div className="icon">📒</div>
                      <div className="label">Contact Me</div>
                    </div>
                    <div className="icon-card">
                      <div className="icon">🛠</div>
                      <div className="label">My Tools</div>
                    </div>
                  </div>
                )}

                {activeSection === "videos" && (
                  <div className="videos-area">
                    <div className="videos-left">
                      <div className="list-title">Playlist</div>
                      {SAMPLE_VIDEOS.map(v => (
                        <div 
                          key={v.id} 
                          className={`playlist-item ${currentVideo?.id === v.id ? "playing":""}`} 
                          onClick={() => setCurrentVideo(v)}
                        >
                          {v.title}
                        </div>
                      ))}
                    </div>

                    <div className="videos-player">
                      {currentVideo ? (
                        <ReactPlayer 
                          url={currentVideo.url} 
                          controls 
                          width="100%" 
                          height="100%"
                        />
                      ) : (
                        <div className="player-empty">Choose a video from the playlist</div>
                      )}
                    </div>
                  </div>
                )}

                {activeSection === "ai" && (
                  <div className="ai-area">
                    <div className="ai-note">AI Chat will appear here (inside BMO screen).</div>
                  </div>
                )}
              </div>
            </div>

            <div className="bmo-body-expanded">
              <div className="bmo-controls-expanded">
                <div className="btn red" />
                <div className="btn blue" />
                <div className="dpad" />
                <div className="btn green" />
              </div>
              <div className="bmo-legs-expanded">
                <div className="leg"><div className="shoe"/></div>
                <div className="leg"><div className="shoe"/></div>
              </div>
            </div>

          </div>
        </div>
      )}
    </div>
  );
}

export default App;
